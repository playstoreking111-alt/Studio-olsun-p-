function rayc(spos,direc,ignore,dist)
    local rai = ray(spos,direc.Unit * dist)
    local rhit,rpos,rrot = workspace:FindPartOnRayWithIgnoreList(rai,ignore,false,false)

function sound(id,vol,pitch,parent,maxdist)
	local mdist = 30 or maxdist
	local newsound = Instance.new("Sound",parent)
	newsound.Volume = vol
	newsound.SoundId = "rbxassetid://"..id
	newsound.Pitch = pitch
	newsound:Play()
	coroutine.resume(coroutine.create(function()
		wait(.1)
		remove(newsound,newsound.TimeLength/newsound.Pitch)
	end))
	return newsound
end

function decompilePart(part,stud)
	stud = stud or 1
	local parts = {}
	if part then
		local point = part.CFrame * cf(part.Size/2) * cf(-v3(stud/2,stud/2,stud/2))
		for i = 0,part.Size.X - stud,stud do
			local p = ins("Part",effects)
			p.Anchored = true
			p.CanCollide = false
			p.Size = v3(stud,stud,stud)
			p.Transparency = part.Transparency
			p.CFrame = point * cf(-i,0,0)
			p.Color = part.Color
			p.Material = part.Material
			table.insert(parts,p)
			for i = 0,part.Size.Y - stud,stud do
				local point = p.CFrame
				local pa = ins("Part",effects)
				pa.Anchored = true
				pa.CanCollide = false
				pa.Size = v3(stud,stud,stud)
				pa.CFrame = point * cf(0,-i,0)
				pa.Color = part.Color
				pa.Transparency = part.Transparency
				pa.Material = part.Material
				table.insert(parts,pa)
				for i = stud,part.Size.Z - stud,stud do
					local point = pa.CFrame
					local p = ins("Part",effects)
					p.Anchored = true
					p.CanCollide = false
					p.Transparency = part.Transparency
					p.Size = v3(stud,stud,stud)
					p.CFrame = point * cf(0,0,-i)
					p.Color = part.Color
					p.Material = part.Material
					table.insert(parts,p)
				end
			end
		end
		return parts
	end
end

function magDamage(pos,partSize,size,Damage,knockback,knockbackTime,knockDelay,damageSound)
	local reg
	local knockPoint
	if typeof(pos) == "Vector3" then
		reg = Region3.new(pos - v3(size/2,size/2,size/2),pos + v3(size/2,size/2,size/2))
		knockPoint = pos
	elseif typeof(pos) == "Instance" then
		knockPoint = pos.Position
		if partSize then
			reg = Region3.new(pos.Position - pos.Size/2,pos.Position + pos.Size/2)
		else
			reg = Region3.new(pos.Position - v3(size/2,size/2,size/2),pos.Position + v3(size/2,size/2,size/2))
		end
	end
	if reg then
		for i,v in pairs(workspace:FindPartsInRegion3WithIgnoreList(reg,{char},250)) do
			local humm = v.Parent:FindFirstChildOfClass("Humanoid")
			if humm then
				damage(humm,Damage,knockback,knockPoint,knockbackTime,knockDelay,damageSound)
			end
		end
	end
end
